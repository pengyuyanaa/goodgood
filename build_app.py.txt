# 飞机大战打包脚本
# 运行依赖：pyinstaller（pip install pyinstaller）
import os
import subprocess

def build_app():
    """打包成桌面可执行文件"""
    # 打包命令（Windows/Mac/Linux通用）
    # --onefile：单文件打包 | --windowed：无控制台窗口 | --name：APP名称
    cmd = [
        "pyinstaller",
        "--onefile",
        "--windowed",
        "--name", "飞机大战",
        "plane_war.py"
    ]

    try:
        print("开始打包APP...")
        subprocess.run(cmd, check=True)
        print("打包完成！可执行文件在 dist 文件夹中。")
        print("Windows：dist/飞机大战.exe")
        print("Mac：dist/飞机大战.app")
        print("Linux：dist/飞机大战")
    except subprocess.CalledProcessError as e:
        print(f"打包失败：{e}")
        print("请先安装pyinstaller：pip install pyinstaller")
    except FileNotFoundError:
        print("未找到pyinstaller，请先执行：pip install pyinstaller")

if __name__ == "__main__":
    build_app()
